# Link this test through the same arena-built llvm-ar object and private
# library closure used by llvm-driver.  Keep LLVM_LINK_COMPONENTS empty: a
# normal LLVMSupport archive in this image could satisfy symbols before the
# driver-private archive and mask an incomplete folded-tool closure.
if(LLVM_DRIVER_PER_INVOCATION_GLOBALS AND LLVM_ENABLE_THREADS AND
   TARGET obj.llvm-ar AND TARGET obj.lld)
  get_property(llvm_ar_lifecycle_runner TARGET obj.llvm-ar
    PROPERTY LLVM_DRIVER_LIFECYCLE_RUNNER)
  get_property(lld_lifecycle_runner TARGET obj.lld
    PROPERTY LLVM_DRIVER_LIFECYCLE_RUNNER)
  if(llvm_ar_lifecycle_runner AND lld_lifecycle_runner)
    function(add_llvm_driver_per_invocation_test)
      set(LLVM_LINK_COMPONENTS)
      set(LLVM_COMMON_DEPENDS)
      set(EXCLUDE_FROM_ALL ON)
      add_llvm_executable(llvm-driver-per-invocation-test
        DISABLE_LLVM_LINK_LLVM_DYLIB
        ${CMAKE_CURRENT_SOURCE_DIR}/tools/llvm-driver/Inputs/per-invocation.cpp
        )
      # Compile the harness against the same rewritten references as the
      # private library closure. This lets it observe that process executors
      # enforce two simultaneous arenas' independently parsed strategies,
      # rather than accidentally testing the ordinary process-global symbol.
      target_compile_options(llvm-driver-per-invocation-test PRIVATE
        "SHELL:-Xclang -fstatic-arena=llvmi_per_invocation_test"
        "SHELL:-Xclang \"-fstatic-arena-list=${LLVM_DRIVER_STATIC_ARENA_LIST}\"")
      llvm_add_driver_producer_list_dependencies(
        llvm-driver-per-invocation-test)
      target_link_libraries(llvm-driver-per-invocation-test PRIVATE
        obj.llvm-ar
        obj.lld)
    endfunction()
    add_llvm_driver_per_invocation_test()

    if((LLVM_ENABLE_PLUGINS OR LLVM_EXPORT_SYMBOLS_FOR_PLUGINS) AND
       LLVM_TOOL_CLANG_BUILD AND
       (NOT (WIN32 OR CYGWIN) OR LLVM_EXPORT_SYMBOLS_FOR_PLUGINS))
      if(WIN32 OR CYGWIN)
        # Let the module name its executable host now.  The selective export
        # list itself is generated by the deferred callback, after obj.clang
        # has completed the helper's private link closure.
        set_property(TARGET llvm-driver-per-invocation-test PROPERTY
          ENABLE_EXPORTS 1)
      endif()
      function(add_llvm_driver_per_invocation_plugin_test)
        set(LLVM_LINK_COMPONENTS)
        add_llvm_library(LLVMDriverPerInvocationPlugin MODULE
          BUILDTREE_ONLY
          DISABLE_LLVM_LINK_LLVM_DYLIB
          PLUGIN_TOOL llvm-driver-per-invocation-test
          ${CMAKE_CURRENT_SOURCE_DIR}/tools/llvm-driver/Inputs/per-invocation-pass-plugin.cpp
          )
        add_llvm_symbol_exports(LLVMDriverPerInvocationPlugin
          ${CMAKE_CURRENT_SOURCE_DIR}/tools/llvm-driver/Inputs/per-invocation-pass-plugin.exports)
      endfunction()
      add_llvm_driver_per_invocation_plugin_test()
      list(APPEND LLVM_TEST_DEPENDS LLVMDriverPerInvocationPlugin)
    endif()

    # Clang is configured after llvm/test in an integrated build. Defer the
    # optional edge to the end of the top-level directory, when both the test
    # helper and obj.clang (including its lifecycle runner) are visible.
    # LLVM-only builds retain the llvm-ar/LLD coverage above.
    function(llvm_configure_driver_per_invocation_clang_test)
      if(NOT TARGET llvm-driver-per-invocation-test)
        return()
      endif()

      if(TARGET obj.clang)
        get_property(clang_lifecycle_runner TARGET obj.clang
          PROPERTY LLVM_DRIVER_LIFECYCLE_RUNNER)
        set(llvm_driver_test_clang_triple)
        if("X86" IN_LIST LLVM_TARGETS_TO_BUILD)
          set(llvm_driver_test_clang_triple x86_64-unknown-linux-gnu)
        elseif("WebAssembly" IN_LIST LLVM_TARGETS_TO_BUILD)
          set(llvm_driver_test_clang_triple wasm32-unknown-unknown)
        endif()
        if(clang_lifecycle_runner AND llvm_driver_test_clang_triple)
          target_link_libraries(llvm-driver-per-invocation-test PRIVATE
            obj.clang)
          target_compile_definitions(llvm-driver-per-invocation-test PRIVATE
            "LLVM_DRIVER_TEST_CLANG_TRIPLE=\"${llvm_driver_test_clang_triple}\"")
        endif()
      endif()

      # On DLL platforms the plugin imports LLVM symbols from the helper's
      # executable image. Export only after the optional obj.clang edge above
      # has made the helper's complete private link closure visible.
      if(TARGET LLVMDriverPerInvocationPlugin)
        export_executable_symbols_for_plugins(
          llvm-driver-per-invocation-test)
        # The helper is the plugin host, not an LLVM library facade.  In the
        # selective-export configuration, export_executable_symbols() exposes
        # non-static direct dependencies through the executable's link
        # interface.  Do not let the plugin pull the folded OBJECT libraries
        # (and a second private LLVM closure) into its own image.
        set_property(TARGET llvm-driver-per-invocation-test PROPERTY
          INTERFACE_LINK_LIBRARIES "")
      endif()
    endfunction()
    cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL
      llvm_configure_driver_per_invocation_clang_test)

    list(APPEND LLVM_TEST_DEPENDS llvm-driver-per-invocation-test)
  endif()
endif()
